// Fungsi Live Fetch menggunakan Public CORS Proxy yang tembus browser async function fetchLiveTwitterAPI() { let rawToken = document.getElementById('apiTokenInput') ? document.getElementById('apiTokenInput').value.trim() : ''; let bearerToken = rawToken.replace(/[^\x00-\x7F]/g, ""); if(!bearerToken) { alert('Mohon masukkan X API v2 Bearer Token milikmu terlebih dahulu melalui menu ⚙️ Settings API di pojok kanan atas!'); openGeneralModal(); return; } const btn = document.querySelector('button[onclick="fetchLiveTwitterAPI()"]'); btn.innerText = '⏳ Menambang dari X...'; btn.disabled = true; try { let queryParam = encodeURIComponent('wtb (netflix OR canva OR capcut OR joki) -is:retweet'); // Menggunakan corsproxy.io untuk melewati blokir CORS browser secara sah let targetUrl = `https://api.twitter.com/2/tweets/search/recent?query=${queryParam}&max_results=10&tweet.fields=created_at,author_id`; let proxyUrl = `https://corsproxy.io/?` + encodeURIComponent(targetUrl); let response = await fetch(proxyUrl, { method: 'GET', headers: { 'Authorization': 'Bearer ' + bearerToken, 'Content-Type': 'application/json' } }); if (!response.ok) { let errDetail = await response.json().catch(() => ({})); throw new Error(errDetail.detail || 'Gagal terhubung ke X API v2. Periksa kembali token atau limit akunmu.'); } let result = await response.json(); if(!result.data || result.data.length === 0) { alert('Pencarian berhasil, namun belum ada tweet WTB terbaru yang cocok pada detik ini.'); btn.innerText = '⚡ Fetch X API v2 (Live)'; btn.disabled = false; return; } let container = document.getElementById('leadsContainer'); result.data.forEach(tweet => { let newCard = document.createElement('div'); newCard.className = 'bg-abyss-purpleDark border border-abyss-crimson rounded-2xl p-5 shadow-lg relative overflow-hidden transition'; newCard.setAttribute('data-status', 'new'); newCard.setAttribute('data-intent', 'inrush'); let dummyUser = '@buyer_x'; newCard.innerHTML = `
Live API 🔥
X

Live Buyer ${dummyUser}

Baru saja • via X Developer API v2

${tweet.text}

Live Lead 🔥 Hot
`; container.prepend(newCard); }); btn.innerText = '⚡ Fetch X API v2 (Live)'; btn.disabled = false; alert('Berhasil menarik data tweet asli secara live dari X Developer API!'); } catch (error) { console.error(error); btn.innerText = '⚡ Fetch X API v2 (Live)'; btn.disabled = false; alert('Gagal mengambil data: ' + error.message); } }
